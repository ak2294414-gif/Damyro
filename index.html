        Table Setup

        Run this SQL in your Supabase SQL editor to create the bookings table if it doesn't exist yet.
        -- 1. Bookings Table (Customer data)
CREATE TABLE IF NOT EXISTS webkrafta_bookings (
  id BIGSERIAL PRIMARY KEY,
  name TEXT NOT NULL,
  biz TEXT NOT NULL,
  phone TEXT NOT NULL,
  email TEXT,
  plan TEXT,
  plan_id INTEGER,
  setup_fee TEXT,
  message TEXT,
  via_wa BOOLEAN DEFAULT true,
  from_mobile BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Settings Table (Admin PIN, global config)
CREATE TABLE IF NOT EXISTS webkrafta_settings (
  key TEXT PRIMARY KEY,
  value TEXT NOT NULL,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Insert default admin PIN (change '2025' if you want a different default)
INSERT INTO webkrafta_settings (key, value) VALUES ('admin_pin', '2025')
ON CONFLICT (key) DO NOTHING;

-- Enable Row Level Security
ALTER TABLE webkrafta_bookings ENABLE ROW LEVEL SECURITY;
ALTER TABLE webkrafta_settings ENABLE ROW LEVEL SECURITY;

-- Allow anonymous inserts (for booking form)
CREATE POLICY "allow_insert_bookings" ON webkrafta_bookings
  FOR INSERT TO anon WITH CHECK (true);

-- Allow anonymous reads (for admin dashboard)
CREATE POLICY "allow_read_bookings" ON webkrafta_bookings
  FOR SELECT TO anon USING (true);

-- Allow anonymous deletes (for admin)
CREATE POLICY "allow_delete_bookings" ON webkrafta_bookings
  FOR DELETE TO anon USING (true);

-- Allow anonymous read/write on settings (needed for PIN sync without login)
CREATE POLICY "allow_all_settings" ON webkrafta_settings
  FOR ALL TO anon USING (true) WITH CHECK (true);
